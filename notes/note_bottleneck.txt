# Fallback in case Traceback isn't perfectly formatted
python_stack_per_tid = collections.defaultdict(list)
events.sort(key=lambda x: x.get('ts', 0))

for e in events:
    name = e.get("name", "")
    tid = e.get("tid", 0)
    ph = e.get("ph", "")
    cat = e.get("cat", "")

    # Track python stack manually as fallback
    if 'python' in cat or '.py' in name:
        if ph == 'B' or ph == 'X':
            python_stack_per_tid[tid].append(name)
        elif ph == 'E' and python_stack_per_tid[tid]:
            python_stack_per_tid[tid].pop()

    if name in ["aten::item", "aten::is_nonzero", "aten::_local_scalar_dense"]:
        args = e.get("args", {})
        tb = None
        for key in ["Python traceback", "Traceback", "Python Line", "Python traceback (most recent call last)"]:
            if key in args:
                tb = args[key]
                break

        if tb is not None:
            if isinstance(tb, list):
                caller = tb[-1]
                for frame in reversed(tb):
                    if 'sglang' in frame or 'fla' in frame:
                        caller = frame
                        break
